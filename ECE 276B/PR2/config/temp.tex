\begin{algorithm}[H]
\caption{DSP via Forward Dynamic Programming}

\KwIn{vertices $\mathcal{V}$, start $s \in \mathcal{V}$, goal $\tau \in \mathcal{V}$, and costs $c_{ij}$ for $i,j \in \mathcal{V}$}
$T = |\mathcal{V}| - 1$\;
\For{$t = 0, \ldots, T$}{$V_t^F(s) = 0$}
\For{$j \in \mathcal{V} \setminus \{\tau\}$}{
    $V_0^F(j) = \infty$\;
    $V_1^F(j) = c_{s,j}$\;
}
\For{$t = 2, \ldots, T$}{
    \For{$j \in \mathcal{V} \setminus \{\tau\}$}{
    $V_t^F(j) = \min\limits_{i \in \mathcal{V}} \left( c_{i,j} + V_{t-1}^F(i) \right)$
    }
    \If{$V_t^F(i) = V_{t-1}^F(i), \ \forall i \in \mathcal{V} \setminus \{\tau\}$}{\textbf{break}}
}

\end{algorithm}
